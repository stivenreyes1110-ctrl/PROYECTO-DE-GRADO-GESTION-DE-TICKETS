/*==============================================================================
  MESA DE AYUDA MULTISERVICIO
  SQL Server - Creacion de tablas, relaciones, indices y parametros de endpoints

  Generado a partir de los endpoints montados en IntranetBk/index.js.

  IMPORTANTE:
  1. Revisar nombres y tipos contra la base existente antes de ejecutar.
  2. El script no elimina ni renombra objetos existentes.
  3. Las tablas se crean solamente cuando no existen.
  4. ARCHIVO_URL admite hasta 1000 caracteres. El controlador actual genera
     una URL completa; para la arquitectura objetivo se recomienda migrarlo
     posteriormente a una ruta relativa /uploads/... sin cambiar esta tabla.
==============================================================================*/

/* Indice principal para Solicitud:
   WHERE IDUSUARIO = @IDUSUARIO AND IDMESA = @IDMESA [AND ESTADO = @ESTADO]
   ORDER BY ID DESC */
IF NOT EXISTS (
    SELECT 1 FROM sys.indexes
    WHERE name = 'IX_TICKETSMESA_USUARIO_MESA_ESTADO_ID'
      AND object_id = OBJECT_ID('dbo.TICKETSMESA')
)
BEGIN
    CREATE INDEX IX_TICKETSMESA_USUARIO_MESA_ESTADO_ID
        ON dbo.TICKETSMESA (IDUSUARIO, IDMESA, ESTADO, ID DESC)
        INCLUDE (
            IDINCIDENTE, IDAREA, IDSEDE, PRIORIDAD,
            HORAREGISTRO, HORACIERRE, IDSOPORTE
        );
END;
GO

  6. PARAMETROS UTILIZADOS POR LOS ENDPOINTS

  Los valores idUsuario, idMesa, idRol e idSoporte marcados como "JWT" deben
  obtenerse del token validado. No deben aceptarse desde el cliente.

/*
POST /api/login
BODY:
  @usuario   VARCHAR(100)  obligatorio
  @password  VARCHAR       obligatorio; Node calcula SHA-256 hexadecimal
  @mesa      INT           obligatorio

CONSULTA DE ACCESO:
  USUARIOSMESA.USUARIO = @usuario
  LEFT JOIN SOPORTESMESA por IDUSUARIO + IDMESA

ROL EFECTIVO:
  SOPORTESMESA.IDROL 1 = ADMIN
  SOPORTESMESA.IDROL 2 = SOPORTE
  Sin relacion en la mesa = USUARIO (rol 3)
*/

/*
GET /api/solicitudes-soporte
JWT:
  @IDUSUARIO INT
  @IDMESA    INT
QUERY:
  @pagina        INT            defecto 1
  @limite        INT            defecto 10; maximo 100
  @estado        INT            1, 2 o 3
  @ticket        NVARCHAR(250)  opcional
  @dependencia   NVARCHAR(250)  opcional
  @solicitud     NVARCHAR(250)  opcional
  @prioridad     NVARCHAR(250)  opcional
  @estadoTexto   NVARCHAR(250)  opcional
  @registro      NVARCHAR(250)  opcional
  @respuesta     NVARCHAR(250)  opcional
  @responsables  NVARCHAR(250)  opcional
SQL PAGINACION:
  @OFFSET INT = (@pagina - 1) * @limite
  @LIMITE INT = @limite
*/

/*
GET /api/solicitudes-soporte/resumen
GET /api/solicitudes-soporte/:id
DELETE /api/solicitudes-soporte/:id
JWT: @IDUSUARIO INT, @IDMESA INT
RUTA: @ID INT
*/

POST /api/solicitudes-soporte
CONTENT-TYPE: multipart/form-data
JWT:
  @IDUSUARIO INT
  @IDMESA INT

  @IDINCIDENTE  INT
  @IDAREA       INT
  @IDSEDE       INT
  @PRIORIDAD    VARCHAR(30): BAJA | MEDIA | ALTA

  @DIRECCION    VARCHAR(100), opcional

SELECT
    t.name AS TABLA,
    SUM(p.rows) AS REGISTROS
FROM sys.tables t
INNER JOIN sys.partitions p
    ON p.object_id = t.object_id
   AND p.index_id IN (0, 1)
WHERE t.name IN (
    'ROLESMESA',
    'MESASAYUDA',
    'SEDESCLB',
    'AREASCLB',
    'INCIDENTESMESA',
    'USUARIOSMESA',
    'SOPORTESMESA',
    'TICKETSMESA',
    'INVENTARIOSMESA',
    'MANTENIMIENTOSMESA',
    'RESPONSABILIDADES'
)
GROUP BY t.name
ORDER BY t.name;
GO

SELECT
    fk.name AS LLAVE_FORANEA,
    OBJECT_NAME(fk.parent_object_id) AS TABLA_HIJA,
    OBJECT_NAME(fk.referenced_object_id) AS TABLA_PADRE
FROM sys.foreign_keys fk
WHERE OBJECT_NAME(fk.parent_object_id) IN (
    'AREASCLB',
    'INCIDENTESMESA',
    'USUARIOSMESA',
    'SOPORTESMESA',
    'TICKETSMESA',
    'INVENTARIOSMESA',
    'MANTENIMIENTOSMESA',
    'RESPONSABILIDADES'
)
ORDER BY TABLA_HIJA, LLAVE_FORANEA;
GO
